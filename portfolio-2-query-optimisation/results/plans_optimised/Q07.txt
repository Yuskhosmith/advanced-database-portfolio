-- Q07 Name prefix search
-- file: sql/queries/q07_student_name_search.sql
-- phase: optimised

SELECT matric_no, first_name, last_name, current_level, status
FROM student
WHERE lower(last_name) LIKE 'okon%' AND lower(first_name) LIKE 'chi%'
ORDER BY matric_no;
